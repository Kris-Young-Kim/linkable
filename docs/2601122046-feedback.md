제공해주신 에러 로그와 데이터베이스 타입을 분석한 결과, 세 가지 주요 문제가 식별되었습니다.

1.  **데이터베이스 스키마 누락 (`PGRST205`)**: `matching_performance_logs`, `icf_iso_precomputed_mappings` 등의 테이블이 Supabase에 생성되지 않았습니다.
2.  **데이터베이스 함수 누락 (`PGRST202`, `0A000`)**: 벡터 검색 및 실시간 학습 관련 함수가 없거나, `FILTER` 구문 오류로 인해 실행되지 않습니다.
3.  **코드와 DB 컬럼 불일치 (`42703`)**: `products` 테이블에서 `iso_code` 컬럼이 삭제되고 `iso_code_id`로 대체되었으나, 코드(`feedback-scorer.ts` 등)에서는 여전히 `iso_code`를 직접 조회하고 있습니다.

이 문제를 해결하기 위해 **1) SQL 마이그레이션 적용**과 **2) TypeScript 코드 수정**이 필요합니다.

### 1단계: Supabase SQL 마이그레이션 실행

Supabase 대시보드의 **SQL Editor**에서 다음 스크립트를 실행하여 누락된 테이블과 함수를 생성하세요.

```sql
-- 1. 매칭 성능 로그 테이블 생성
CREATE TABLE IF NOT EXISTS public.matching_performance_logs (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    consultation_id UUID REFERENCES public.consultations(id) ON DELETE SET NULL,
    user_id UUID REFERENCES public.users(id) ON DELETE SET NULL,
    weight_config_id UUID, -- matching_weight_configs가 있다면 REFERENCES 추가
    weight_config_name TEXT,
    icf_codes TEXT[] NOT NULL,
    icf_code_count INTEGER,
    matched_iso_codes TEXT[],
    match_count INTEGER,
    top_match_score NUMERIC,
    average_match_score NUMERIC,
    execution_time_ms INTEGER,
    semantic_match_used BOOLEAN DEFAULT false,
    knowledge_graph_used BOOLEAN DEFAULT false,
    recommendation_clicked BOOLEAN DEFAULT false,
    purchase_completed BOOLEAN DEFAULT false,
    feedback_rating INTEGER
);

-- 2. 사전 계산된 매핑 테이블 생성
CREATE TABLE IF NOT EXISTS public.icf_iso_precomputed_mappings (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    last_used_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
    icf_codes_key TEXT NOT NULL UNIQUE,
    icf_codes TEXT[] NOT NULL,
    iso_matches JSONB NOT NULL, -- IsoMatch[] 구조 저장
    match_method TEXT,
    confidence_score NUMERIC DEFAULT 0,
    usage_count INTEGER DEFAULT 0,
    success_rate NUMERIC DEFAULT 0
);

-- 3. 실시간 학습 통계 테이블 (get_realtime_weight_adjustment 함수용)
CREATE TABLE IF NOT EXISTS public.realtime_learning_stats (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    icf_codes_key TEXT,
    iso_code TEXT,
    weight_adjustment NUMERIC DEFAULT 1.0,
    click_rate NUMERIC DEFAULT 0,
    impression_count INTEGER DEFAULT 0,
    last_seen_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- 4. 누락된 함수 생성: get_realtime_weight_adjustment
CREATE OR REPLACE FUNCTION public.get_realtime_weight_adjustment(
    p_icf_codes text[],
    p_iso_code text
)
RETURNS numeric
LANGUAGE plpgsql
AS $$
DECLARE
    v_adjustment numeric;
    v_icf_key text;
BEGIN
    -- ICF 코드 배열을 정렬하여 키 생성
    SELECT string_agg(x, ',') INTO v_icf_key
    FROM (SELECT unnest(p_icf_codes) x ORDER BY 1) t;

    SELECT weight_adjustment INTO v_adjustment
    FROM public.realtime_learning_stats
    WHERE icf_codes_key = v_icf_key AND iso_code = p_iso_code;

    RETURN COALESCE(v_adjustment, 1.0);
END;
$$;

-- 5. 누락된 함수 생성: search_similar_icf_iso_mappings
-- (pgvector 확장이 필요합니다. 없다면 'CREATE EXTENSION IF NOT EXISTS vector;' 실행)
CREATE OR REPLACE FUNCTION public.search_similar_icf_iso_mappings(
    query_embedding vector(768),
    similarity_threshold float,
    max_results int
)
RETURNS TABLE (
    id uuid,
    icf_codes text[],
    iso_code text,
    iso_label text,
    iso_description text,
    base_score float,
    usage_count int,
    success_rate float,
    similarity float
)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT
        e.id,
        e.icf_codes,
        e.iso_code,
        e.iso_label,
        e.iso_description,
        CAST(e.base_score AS float),
        e.usage_count,
        CAST(e.success_rate AS float),
        CAST(1 - (e.embedding <=> query_embedding) AS float) as similarity
    FROM
        public.icf_iso_embeddings e
    WHERE
        1 - (e.embedding <=> query_embedding) > similarity_threshold
    ORDER BY
        similarity DESC
    LIMIT max_results;
END;
$$;

-- 6. 인덱스 생성
CREATE INDEX IF NOT EXISTS idx_matching_perf_consultation ON public.matching_performance_logs(consultation_id);
CREATE INDEX IF NOT EXISTS idx_precomputed_mappings_key ON public.icf_iso_precomputed_mappings(icf_codes_key);
```
