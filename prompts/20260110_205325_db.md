권장 데이터베이스 변경

1. ICF-ISO 매핑 테이블 생성 (핵심)

-- ICF 코드와 ISO 코드 간의 매핑 관계
CREATE TABLE icf_iso_mappings (
id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

-- ICF 코드 (여러 개 가능)
icf_codes TEXT[] NOT NULL,                    -- 예: ['d450', 'd465']

-- ISO 코드 (Subclass 레벨)
iso_code VARCHAR(10) NOT NULL,                -- 예: '12 06', '15 09'
iso_code_id UUID REFERENCES iso_codes(id),    -- FK (선택적)

-- 매핑 정보
label VARCHAR(100) NOT NULL,                  -- 예: '보행 보조기기'
description TEXT,                             -- 상세 설명
base_score NUMERIC(3,2) DEFAULT 0.70,         -- 매칭 기본 점수 (0-1)

-- 메타데이터
source VARCHAR(50) DEFAULT 'manual',          -- 'manual', 'ai_inferred', 'research'
confidence NUMERIC(3,2) DEFAULT 1.0,          -- 신뢰도
is_active BOOLEAN DEFAULT true,
created_at TIMESTAMPTZ DEFAULT now(),
updated_at TIMESTAMPTZ DEFAULT now()
);

-- 인덱스
CREATE INDEX idx_icf_iso_mappings_icf ON icf_iso_mappings USING GIN(icf_codes);
CREATE INDEX idx_icf_iso_mappings_iso ON icf_iso_mappings(iso_code);

2. Products 테이블 정규화

-- 기존 iso_code 문자열을 iso_code_id FK로 마이그레이션
UPDATE products p
SET iso_code_id = ic.id
FROM iso_codes ic
WHERE p.iso_code = ic.code
AND p.iso_code_id IS NULL;

-- 이후 iso_code 컬럼은 deprecated 처리 (기존 호환성 유지)
COMMENT ON COLUMN products.iso_code IS
'DEPRECATED: iso_code_id FK 사용 권장. 마이그레이션 후 제거 예정';
